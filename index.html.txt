<button onclick="spin()">QUAY</button>

<script>
// ảnh DEMO (sẽ thay bằng ảnh mạt chược thật)
const imgs = [
  "https://dummyimage.com/80x80/ffffff/000000.png&text=A",
  "https://dummyimage.com/80x80/ffffff/000000.png&text=B",
  "https://dummyimage.com/80x80/ffffff/000000.png&text=C",
  "https://dummyimage.com/80x80/ffeb3b/000000.png&text=W", // wild
  "https://dummyimage.com/80x80/ff9800/000000.png&text=S"  // scatter
];

const board = document.getElementById("board");
let cells = [];

for(let i=0;i<25;i++){
  let d=document.createElement("div");
  d.className="tile";
  let img=document.createElement("img");
  d.appendChild(img);
  board.appendChild(d);
  cells.push(img);
}

function spin(){
  let col=0;
  let timer=setInterval(()=>{
    for(let r=0;r<5;r++){
      let idx = col + r*5;
      cells[idx].src = imgs[Math.floor(Math.random()*imgs.length)];
    }
    col++;
    if(col>=5) clearInterval(timer);
  },300); // delay từng cột
}
</script>
